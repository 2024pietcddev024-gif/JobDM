27: function calculateSimilarity(str1, str2) {
28:   const words1 = new Set(str1.toLowerCase().replace(/[^\w\s]/g, '').split(/\s+/).filter(w => w.length > 3));
29:   const words2 = new Set(str2.toLowerCase().replace(/[^\w\s]/g, '').split(/\s+/).filter(w => w.length > 3));
30:   let intersection = 0;
31:   for (let w of words1) if (words2.has(w)) intersection++;
32:   const union = words1.size + words2.size - intersection;
33:   return union === 0 ? 0 : intersection / union;
